cmake_minimum_required(VERSION 3.15.2)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(GalaxCoreRegressionPlatformGui LANGUAGES CXX)
  set(CMAKE_CXX_STANDARD 17)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  set(CMAKE_CXX_EXTENSIONS OFF)
  set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
  include(CTest)
endif()
find_package(Qt5 5.8 REQUIRED COMPONENTS Core Widgets Network)
add_library(gcrp_qt_transport STATIC src/RpcClient.cpp)
target_include_directories(gcrp_qt_transport PUBLIC
  ${CMAKE_CURRENT_SOURCE_DIR}/include ${CMAKE_CURRENT_SOURCE_DIR}/../../third_party)
target_link_libraries(gcrp_qt_transport PUBLIC Qt5::Core Qt5::Network)
add_executable(gcrp_gui src/main.cpp src/MainWindow.cpp src/ConsolePages.cpp
  src/TaskViews.cpp src/WorkerViews.cpp src/DailyView.cpp src/Review.cpp)
target_link_libraries(gcrp_gui PRIVATE gcrp_qt_transport Qt5::Widgets)
if(MSVC)
  target_compile_options(gcrp_gui PRIVATE /utf-8)
endif()
if(BUILD_TESTING)
  add_executable(test_qt_archive_download tests/archive_download.cpp)
  target_link_libraries(test_qt_archive_download PRIVATE gcrp_qt_transport)
  add_test(NAME test_qt_archive_download COMMAND test_qt_archive_download)
  set_tests_properties(test_qt_archive_download PROPERTIES TIMEOUT 120)
  add_executable(test_qt_transport tests/transport.cpp)
  target_link_libraries(test_qt_transport PRIVATE gcrp_qt_transport)
  add_test(NAME test_qt_transport COMMAND test_qt_transport)
  add_test(NAME test_qt_gui COMMAND gcrp_gui --demo --self-test)
  add_test(NAME test_qt_review COMMAND gcrp_gui --demo --review-dir ${CMAKE_CURRENT_BINARY_DIR}/review-v3)
  set_tests_properties(test_qt_review PROPERTIES TIMEOUT 30 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  set_tests_properties(test_qt_transport test_qt_gui PROPERTIES TIMEOUT 15)
  set_tests_properties(test_qt_gui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  find_package(Python3 3.6 QUIET COMPONENTS Interpreter)
  if(Python3_Interpreter_FOUND)
    add_test(NAME test_qt_cli COMMAND ${Python3_EXECUTABLE} -B
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli.py $<TARGET_FILE:gcrp_gui>)
    set_tests_properties(test_qt_cli PROPERTIES TIMEOUT 45)
    add_test(NAME test_qt_catalog COMMAND ${Python3_EXECUTABLE} -B
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/catalog.py
      ${CMAKE_CURRENT_SOURCE_DIR}/../../tools/export_gui_catalog.py)
    set_tests_properties(test_qt_catalog PROPERTIES TIMEOUT 15)
  endif()
endif()
